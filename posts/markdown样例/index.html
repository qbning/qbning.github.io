    var sum = 0;

    return {
        add: function() {
            sum += 1;
            return sum;
        },
        reset: function() {
            return sum = 0;
        }
    };
}());

alert(模块模式.add());    // 弹出: 1
alert(模块模式.add());    // 弹出: 2
alert(模块模式.reset());  // 弹出: 0

分隔线#

在 HTML 中 <hr> 是用来创建段落之间的分隔线。在 Markdown 中，你可以通过以下任意一种方式创建：

  • ___：三个下划线
  • ---：三个短横线
  • ***：三个星号
